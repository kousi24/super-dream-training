public class Main {
    public static void main(String[] args) {

        int[] arr = {10, 40, 20, 70, 30};

        int max = arr[0];

        for (int i = 1; i < arr.length; i++) {

            if (arr[i] > max) {
                max = arr[i];
            }
        }

        System.out.println(max);
    }
}
